cmake_minimum_required(VERSION 3.20)
project(PulseStream VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(PULSE_SANITIZER "" CACHE STRING "address or thread (separate builds)")
if(PULSE_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(PULSE_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
find_package(PostgreSQL REQUIRED)
find_package(OpenSSL REQUIRED)
include(FetchContent)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/json/CMakeLists.txt")
  add_subdirectory(third_party/json EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
  FetchContent_MakeAvailable(json)
endif()
# Upstream uses CP936 source text. Transcode build copies, preserving original bytes.
find_program(ICONV_EXECUTABLE iconv REQUIRED)
foreach(source ddq.c test_ddq.c)
  execute_process(COMMAND ${ICONV_EXECUTABLE} -f CP936 -t UTF-8 ${CMAKE_CURRENT_SOURCE_DIR}/${source}
    OUTPUT_FILE ${CMAKE_CURRENT_BINARY_DIR}/utf8_${source} COMMAND_ERROR_IS_FATAL ANY)
endforeach()
add_library(ddq STATIC ${CMAKE_CURRENT_BINARY_DIR}/utf8_ddq.c)
if(APPLE)
  target_compile_definitions(ddq PRIVATE _DARWIN_C_SOURCE)
endif()
# Do not put the repository root on -I: VERSION shadows C++ <version> on macOS.
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/upstream_include)
execute_process(COMMAND ${ICONV_EXECUTABLE} -f CP936 -t UTF-8 ${CMAKE_CURRENT_SOURCE_DIR}/ddq.h
  OUTPUT_FILE ${CMAKE_CURRENT_BINARY_DIR}/upstream_include/ddq.h COMMAND_ERROR_IS_FATAL ANY)
target_include_directories(ddq PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/upstream_include)
add_library(pulse STATIC src/model.cpp src/queue.cpp src/database.cpp src/service.cpp src/http.cpp)
target_include_directories(pulse PUBLIC include)
target_link_libraries(pulse PUBLIC ddq nlohmann_json::nlohmann_json PostgreSQL::PostgreSQL OpenSSL::Crypto Threads::Threads)
target_compile_options(pulse PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(pulse PUBLIC PULSE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(pulsestream src/main.cpp)
target_link_libraries(pulsestream PRIVATE pulse)
add_executable(device_simulator src/simulator.cpp)
target_link_libraries(device_simulator PRIVATE pulse)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.17.0)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(pulse_tests tests/unit.cpp)
  target_link_libraries(pulse_tests PRIVATE pulse GTest::gtest_main)
  include(GoogleTest)
  if(PULSE_SANITIZER)
    add_test(NAME pulse_sanitized COMMAND pulse_tests)
    set_tests_properties(pulse_sanitized PROPERTIES TIMEOUT 60)
  else()
    gtest_discover_tests(pulse_tests DISCOVERY_TIMEOUT 30)
  endif()
  add_executable(ddq_upstream_tests ${CMAKE_CURRENT_BINARY_DIR}/utf8_test_ddq.c)
  if(APPLE)
    target_compile_definitions(ddq_upstream_tests PRIVATE _DARWIN_C_SOURCE)
  endif()
  target_link_libraries(ddq_upstream_tests PRIVATE ddq)
  add_test(NAME upstream_ddq COMMAND ddq_upstream_tests)
  set_tests_properties(upstream_ddq PROPERTIES TIMEOUT 30)
endif()
